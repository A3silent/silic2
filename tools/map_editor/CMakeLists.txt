cmake_minimum_required(VERSION 3.16)
project(silic2_map_editor LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SILIC2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Silic2 game source directory (shared Map / JSON)")
get_filename_component(SILIC2_ROOT "${SILIC2_ROOT}" ABSOLUTE)
if(NOT EXISTS "${SILIC2_ROOT}/include/engine/map.h")
    message(FATAL_ERROR "Set -DSILIC2_ROOT=/path/to/silic2; shared engine/map.h was not found")
endif()
find_package(OpenGL REQUIRED)
find_package(glfw3 REQUIRED)
add_library(silic2_map_data STATIC "${SILIC2_ROOT}/src/engine/map.cpp" "${SILIC2_ROOT}/src/engine/simple_json.cpp" "${SILIC2_ROOT}/src/engine/mesh_asset.cpp" "${SILIC2_ROOT}/src/engine/spatial_world.cpp")
target_include_directories(silic2_map_data PUBLIC "${SILIC2_ROOT}/include" "${SILIC2_ROOT}/include/glm")
add_library(editor_document STATIC src/editor_document.cpp)
target_include_directories(editor_document PUBLIC include)
target_link_libraries(editor_document PUBLIC silic2_map_data)
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/imgui")
add_library(imgui STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/backends/imgui_impl_glfw.cpp ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui PUBLIC "${IMGUI_DIR}" "${IMGUI_DIR}/backends")
target_link_libraries(imgui PUBLIC OpenGL::GL glfw)
add_executable(map_editor src/map_editor_final.cpp tests/editor_self_test.cpp)
target_link_libraries(map_editor PRIVATE editor_document imgui OpenGL::GL glfw)
target_compile_definitions(map_editor PRIVATE GL_SILENCE_DEPRECATION SILIC2_ROOT="${SILIC2_ROOT}" SILIC2_EDITOR_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
if(NOT APPLE)
    target_sources(map_editor PRIVATE "${SILIC2_ROOT}/src/engine/glad.c")
endif()
enable_testing()
add_test(NAME editor_document_roundtrip COMMAND map_editor --self-test "${SILIC2_ROOT}/res/maps/test_room.json" "${CMAKE_CURRENT_BINARY_DIR}/test-output")
add_test(NAME editor_textured_legacy COMMAND map_editor --self-test "${SILIC2_ROOT}/res/maps/textured_room.json" "${CMAKE_CURRENT_BINARY_DIR}/test-output-textured")
add_test(NAME editor_arena_v2 COMMAND map_editor --self-test "${SILIC2_ROOT}/tests/fixtures/starry_crystal_arena_v2.json" "${CMAKE_CURRENT_BINARY_DIR}/test-output-arena")

add_test(NAME editor_hollow_orbit COMMAND map_editor --self-test "${SILIC2_ROOT}/res/maps/hollow_orbit_arena.json" "${CMAKE_CURRENT_BINARY_DIR}/test-output-hollow-orbit")

add_test(NAME editor_exploration COMMAND map_editor --self-test "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/crystal_observatory.json" "${CMAKE_CURRENT_BINARY_DIR}/test-output-exploration")

add_test(NAME editor_hybrid_geometry COMMAND map_editor --hybrid-self-test "${CMAKE_CURRENT_BINARY_DIR}/test-output-hybrid")

add_test(NAME editor_crystal_combat COMMAND map_editor --self-test "${SILIC2_ROOT}/res/maps/starry_crystal_arena.json" "${CMAKE_CURRENT_BINARY_DIR}/test-output-crystal-combat")
